#include <stdio.h>
#include <stdlib.h>

void swap (int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void selectionSort(int arr[], int n) {
    for(int i = 0; i< n; i++) {
	int minIndex = i;
	
	for( int j = i + 1; j<n; j++) {
		if(arr[j] < arr[minIndex]) {
			minIndex = j;
		}
	}
	swap(&arr[minIndex], &arr[i]);
    }
}

void testSort() {
    int n = 5;
    int arr[] = {64, 25, 12, 22, 11};
    selectionSort(arr, n);
}

int main () {
    testSort();
}
